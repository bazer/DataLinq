using System.Diagnostics;
using System.Reflection;
using System.Runtime.InteropServices;
using System.Runtime.Loader;
using System.Security.Cryptography;
using System.Text.Json;

// Supplemental descriptive timing only. Delegate to the actual harness methods;
// do not multiply invocations inside one cold-cache/transaction setup.
if (args.Length != 4) throw new ArgumentException("Usage: benchmark.dll stages|reads|mutations forward|reverse fresh-output.json");
var assemblyPath = Path.GetFullPath(args[0]);
var group = args[1];
var reverse = args[2] switch { "forward" => false, "reverse" => true, _ => throw new ArgumentException("Invalid order") };
var output = Path.GetFullPath(args[3]);
if (File.Exists(output)) throw new IOException("Output exists");
var resolver = new AssemblyDependencyResolver(assemblyPath);
AssemblyLoadContext.Default.Resolving += (context, name) => resolver.ResolveAssemblyToPath(name) is { } path ? context.LoadFromAssemblyPath(path) : null;
AssemblyLoadContext.Default.ResolvingUnmanagedDll += (_, name) => resolver.ResolveUnmanagedDllToPath(name) is { } path ? NativeLibrary.Load(path) : IntPtr.Zero;
var assembly = AssemblyLoadContext.Default.LoadFromAssemblyPath(assemblyPath);
var recipes = group switch
{
    "stages" => new[]
    {
        new Recipe("EmployeesBenchmarks", "BinaryCanonicalKeyPropagation", 1000),
        new Recipe("EmployeesBenchmarks", "TypedIdCanonicalKeyPropagation", 1000),
        new Recipe("EmployeesBenchmarks", "CanonicalProviderRowDecoding", 1000),
        new Recipe("EmployeesBenchmarks", "KnownMissMaterializationPublication", 1000, Setup: true),
        new Recipe("EmployeesBenchmarks", "MutationCommandPreparation", 1000, Setup: true, Cleanup: true),
        new Recipe("EmployeesBenchmarks", "MutationExecutionPreflight", 1000, Setup: true, Cleanup: true),
        new Recipe("EmployeesBenchmarks", "MutationFinalDriftValidation", 1000, Setup: true, Cleanup: true),
        new Recipe("EmployeesBenchmarks", "MutationStateChangeCapture", 1000),
        new Recipe("EmployeesBenchmarks", "SourceCacheResultPublication", 1000),
        new Recipe("EmployeesBenchmarks", "SourceResultValidation", 1000),
        new Recipe("EmployeesBenchmarks", "WarmPrimaryKeyFetch", 1000, Setup: true)
    },
    "reads" => new[]
    {
        new Recipe("EmployeesBenchmarks", "RepeatedInPredicateFetch", 1000, Setup: true),
        new Recipe("EmployeesBenchmarks", "RepeatedNonPrimaryKeyEqualityFetch", 1000, Setup: true),
        new Recipe("TypedIdExactTerminalBenchmarks", "ColdTypedIdExactTerminal", 1000, Setup: true),
        new Recipe("TypedIdExactTerminalBenchmarks", "WarmTypedIdExactTerminal", 1000, Setup: true)
    },
    "mutations" => new[]
    {
        new Recipe("AllocationRegressionBenchmarks", "UpdateEmployees", 2000, Setup: true, Cleanup: true),
        new Recipe("AllocationRegressionBenchmarks", "CrudWorkflowSmall", 350, Setup: true, Cleanup: true),
        new Recipe("AllocationRegressionBenchmarks", "CrudWorkflowBatch", 350, Setup: true, Cleanup: true)
    },
    _ => throw new ArgumentException("Invalid group")
};
var providers = group == "mutations" ? new[] { "sqlite-memory" } : new[] { "sqlite-file", "sqlite-memory" };
var cases = recipes.SelectMany(recipe => providers.Select(provider => (Recipe: recipe, Provider: provider))).ToArray();
if (reverse) Array.Reverse(cases);
const int minimumCalls = 64, maximumCalls = 1000000;
const double minimumWarmupSeconds = 2, minimumMeasuredSeconds = 0.25;
var rows = new List<object>();
var dependencies = new Dictionary<string, DependencyIdentity>(StringComparer.OrdinalIgnoreCase);
var startedAt = DateTime.UtcNow;
var assemblyHash = Hash(assemblyPath);
foreach (var (recipe, provider) in cases)
{
    var type = assembly.GetType("DataLinq.Benchmark." + recipe.Type, throwOnError: true)!;
    var benchmark = Activator.CreateInstance(type)!;
    type.GetProperty("ProviderName")!.SetValue(benchmark, provider);
    var method = type.GetMethod(recipe.Method) ?? throw new MissingMethodException(recipe.Method);
    var attribute = method.GetCustomAttributes().Single(value => value.GetType().FullName == "BenchmarkDotNet.Attributes.BenchmarkAttribute");
    if ((int)attribute.GetType().GetProperty("OperationsPerInvoke")!.GetValue(attribute)! != recipe.Operations)
        throw new InvalidDataException("Operation count mismatch");
    var globalSetup = RequiredAction("GlobalSetup");
    var globalCleanup = RequiredAction("GlobalCleanup");
    var setup = IterationAction("Setup", recipe.Setup);
    var cleanup = IterationAction("Cleanup", recipe.Cleanup);
    var work = method.CreateDelegate<Func<int>>(benchmark);
    // Preallocate all measurement storage outside setup and timing.
    var ticks = new long[maximumCalls];
    var bytes = new long[maximumCalls];
    var warmupTicks = new long[maximumCalls];
    var warmupBytes = new long[maximumCalls];
    var warmupChecksums = new int[maximumCalls];
    var measuredChecksums = new int[maximumCalls];
    globalSetup();
    try
    {
        RememberDependencies();
        int? expectedChecksum = null;
        // Each CRUD invocation inserts and deletes 350 rows. SQLite AUTOINCREMENT
        // advances every inserted ID by 350 on the next invocation, and the
        // checksum sums all 350 generated IDs. Cleanup only restores updates.
        var checksumIncrement = recipe.Method is "CrudWorkflowSmall" or "CrudWorkflowBatch"
            ? checked(recipe.Operations * recipe.Operations) : 0;
        // Binary keys use System.HashCode's process-randomized seed. Validate the
        // audited fixture formula inside each process, not numeric equality across processes.
        if (recipe.Method == "BinaryCanonicalKeyPropagation")
        {
            var keyType = AppDomain.CurrentDomain.GetAssemblies().Single(value => value.GetName().Name == "DataLinq")
                .GetType("DataLinq.Instances.DataLinqKey", throwOnError: true)!;
            var keyFactory = keyType.GetMethods(BindingFlags.Public | BindingFlags.Static)
                .Single(value => value.Name == "FromValue" && value.IsGenericMethodDefinition && value.GetGenericArguments().Length == 1 && value.GetParameters().Length == 1)
                .MakeGenericMethod(typeof(byte[]));
            var key = keyFactory.Invoke(null, [Enumerable.Range(0, 16).Select(value => (byte)value).ToArray()])!;
            expectedChecksum = unchecked(4 * recipe.Operations * key.GetHashCode());
        }
        var warmupStart = Stopwatch.GetTimestamp();
        var warmups = 0;
        do
        {
            if (warmups == maximumCalls) throw new InvalidDataException("Warmup limit reached");
            setup();
            var allocatedBefore = GC.GetAllocatedBytesForCurrentThread();
            var start = Stopwatch.GetTimestamp();
            var result = work();
            var end = Stopwatch.GetTimestamp();
            warmupBytes[warmups] = GC.GetAllocatedBytesForCurrentThread() - allocatedBefore;
            warmupTicks[warmups] = end - start;
            warmupChecksums[warmups] = result;
            cleanup();
            expectedChecksum ??= result;
            if (result != unchecked(expectedChecksum.Value + warmups * checksumIncrement))
                throw new InvalidDataException($"Unexpected warmup checksum at invocation {warmups}: {result}");
            warmups++;
        } while (warmups < minimumCalls || Stopwatch.GetElapsedTime(warmupStart).TotalSeconds < minimumWarmupSeconds);
        var warmupElapsedSeconds = Stopwatch.GetElapsedTime(warmupStart).TotalSeconds;
        var measured = 0;
        long totalTicks = 0;
        do
        {
            if (measured == maximumCalls) throw new InvalidDataException("Measurement limit reached");
            setup();
            var allocatedBefore = GC.GetAllocatedBytesForCurrentThread();
            var start = Stopwatch.GetTimestamp();
            var result = work();
            var end = Stopwatch.GetTimestamp();
            bytes[measured] = GC.GetAllocatedBytesForCurrentThread() - allocatedBefore;
            ticks[measured] = end - start;
            measuredChecksums[measured] = result;
            totalTicks += end - start;
            cleanup();
            if (result != unchecked(expectedChecksum.Value + (warmups + measured) * checksumIncrement))
                throw new InvalidDataException($"Unexpected measured checksum at invocation {warmups + measured}: {result}");
            measured++;
        } while (measured < minimumCalls || (double)totalTicks / Stopwatch.Frequency < minimumMeasuredSeconds);
        var scenario = type.GetField("executedScenario", BindingFlags.NonPublic | BindingFlags.Instance)!.GetValue(benchmark)
            ?? throw new InvalidDataException("No scenario selected");
        object telemetryOwner = benchmark;
        if (recipe.Type != "AllocationRegressionBenchmarks")
            telemetryOwner = type.GetField("context", BindingFlags.NonPublic | BindingFlags.Instance)!.GetValue(benchmark)!;
        var telemetry = telemetryOwner.GetType().GetMethod("CaptureTelemetryDelta", BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.Instance)!
            .Invoke(telemetryOwner, [scenario, provider])!;
        RememberDependencies();
        var invocationTicks = ticks[..measured];
        var invocationBytes = bytes[..measured];
        var usPerOperation = invocationTicks.Select(value => value * 1_000_000d / Stopwatch.Frequency / recipe.Operations).ToArray();
        var sorted = usPerOperation.Order().ToArray();
        var mean = usPerOperation.Average();
        rows.Add(new
        {
            recipe.Type, recipe.Method, Provider = provider, OperationsPerInvoke = recipe.Operations,
            Setup = recipe.Setup ? "Setup" + recipe.Method : null, Cleanup = recipe.Cleanup ? "Cleanup" + recipe.Method : null,
            Warmups = warmups, WarmupElapsedSeconds = warmupElapsedSeconds,
            WarmupInvocationTicks = warmupTicks[..warmups], WarmupInvocationBytes = warmupBytes[..warmups],
            MeasuredInvocations = measured, TotalMeasuredWorkSeconds = (double)totalTicks / Stopwatch.Frequency,
            InvocationTicks = invocationTicks, InvocationBytes = invocationBytes,
            InitialChecksum = expectedChecksum.Value, ChecksumIncrementPerInvocation = checksumIncrement,
            WarmupInvocationChecksums = warmupChecksums[..warmups], InvocationChecksums = measuredChecksums[..measured],
            ChecksumRule = checksumIncrement != 0 ? "unchecked(InitialChecksum + zeroBasedInvocation * 350 * 350); SQLite AUTOINCREMENT advances every generated ID; initial checksum equal across runs"
                : recipe.Method == "BinaryCanonicalKeyPropagation" ? "unchecked(4 * OperationsPerInvoke * DataLinqKey.FromValue(bytes 00..0f).GetHashCode()); process-local hash" : "Stable integer checksum; equal across runs",
            MeanMicrosecondsPerOperation = mean, MedianMicrosecondsPerOperation = (sorted[(measured - 1) / 2] + sorted[measured / 2]) / 2,
            MinimumMicrosecondsPerOperation = sorted[0], MaximumMicrosecondsPerOperation = sorted[^1],
            StandardDeviationMicrosecondsPerOperation = Math.Sqrt(usPerOperation.Sum(value => (value - mean) * (value - mean)) / (measured - 1)),
            ManagedThreadBytesPerOperation = (double)invocationBytes.Sum() / measured / recipe.Operations,
            TelemetryReplay = JsonSerializer.SerializeToElement(telemetry, telemetry.GetType())
        });
        Console.WriteLine($"{recipe.Method}/{provider}: {mean:F4} us/op; {measured} measured calls; {totalTicks * 1000d / Stopwatch.Frequency:F1} ms accumulated work");
    }
    finally { globalCleanup(); }
    RememberDependencies();

    Action RequiredAction(string name) => type.GetMethod(name)?.CreateDelegate<Action>(benchmark) ?? throw new MissingMethodException(type.FullName, name);
    Action IterationAction(string kind, bool required)
    {
        var candidates = type.GetMethods().Where(m => m.GetCustomAttributes().Any(a =>
            a.GetType().FullName == "BenchmarkDotNet.Attributes.Iteration" + kind + "Attribute" &&
            a.GetType().GetProperty("Targets")?.GetValue(a) is string[] targets &&
            (targets.Length == 0 || targets.Contains(recipe.Method)))).ToArray();
        if (candidates.Length != (required ? 1 : 0) || required && candidates[0].Name != kind + recipe.Method)
            throw new InvalidDataException("Unexpected iteration lifecycle: " + recipe.Method);
        return required ? candidates[0].CreateDelegate<Action>(benchmark) : static () => { };
    }
}
RememberDependencies();
if (Hash(assemblyPath) != assemblyHash) throw new InvalidDataException("Benchmark assembly changed");
var report = new
{
    Purpose = "Supplemental descriptive timing and thread-allocation control, not BenchmarkDotNet or strict release evidence. No sample trimming or timing-overhead subtraction.",
    Protocol = "Real harness delegates; setup/cleanup outside every measured invocation; cold state reset for every invocation; fixed minimum 64 warmup calls and two elapsed warmup seconds, then minimum 64 measured calls and 250 ms accumulated measured work; maximum 1000000 calls per phase; no forced GC.",
    Group = group, Order = args[2], StartedAtUtc = startedAt, CompletedAtUtc = DateTime.UtcNow,
    Runtime = RuntimeInformation.FrameworkDescription, Architecture = RuntimeInformation.ProcessArchitecture.ToString(),
    StopwatchFrequency = Stopwatch.Frequency, MinimumCalls = minimumCalls, MaximumCalls = maximumCalls, MinimumWarmupSeconds = minimumWarmupSeconds, MinimumMeasuredSeconds = minimumMeasuredSeconds,
    Assembly = assemblyPath, AssemblySha256 = assemblyHash, AssemblyVersion = assembly.GetCustomAttribute<AssemblyInformationalVersionAttribute>()?.InformationalVersion,
    ProbeAssemblySha256 = Hash(Assembly.GetExecutingAssembly().Location), Dependencies = dependencies.Values.OrderBy(value => value.Path).ToArray(), Rows = rows
};
File.WriteAllText(output, JsonSerializer.Serialize(report, new JsonSerializerOptions { WriteIndented = true }));

void RememberDependencies()
{
    var directory = Path.GetDirectoryName(assemblyPath)! + Path.DirectorySeparatorChar;
    foreach (var dependency in AppDomain.CurrentDomain.GetAssemblies().Where(value => !value.IsDynamic && !string.IsNullOrEmpty(value.Location) && value.Location.StartsWith(directory, StringComparison.OrdinalIgnoreCase)))
    {
        var identity = new DependencyIdentity(dependency.GetName().Name!, dependency.Location,
            dependency.GetCustomAttribute<AssemblyInformationalVersionAttribute>()?.InformationalVersion,
            dependency.GetCustomAttributes<AssemblyMetadataAttribute>().ToDictionary(value => value.Key, value => value.Value),
            dependency.ManifestModule.ModuleVersionId, Hash(dependency.Location));
        if (dependencies.TryGetValue(identity.Path, out var prior) && prior.Sha256 != identity.Sha256)
            throw new InvalidDataException("Dependency changed: " + identity.Path);
        dependencies[identity.Path] = identity;
    }
}
static string Hash(string path) => Convert.ToHexString(SHA256.HashData(File.ReadAllBytes(path))).ToLowerInvariant();
sealed record Recipe(string Type, string Method, int Operations, bool Setup = false, bool Cleanup = false);
sealed record DependencyIdentity(string Name, string Path, string? InformationalVersion, Dictionary<string, string?> Metadata, Guid ModuleVersionId, string Sha256);
